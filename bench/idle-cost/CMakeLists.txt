cmake_minimum_required(VERSION 3.20)
project(http_idle_cost LANGUAGES C CXX)
add_subdirectory(../.. http-deps)
find_package(OpenSSL 3 REQUIRED)
add_executable(http_idle_cost server.cc)
target_link_libraries(http_idle_cost PRIVATE http OpenSSL::SSL)
add_executable(http_idle_owners server.cc)
target_compile_definitions(http_idle_owners PRIVATE HTTP_IDLE_SSL_OWNERS=1 TASK_COUNT_FRAMES=1)
target_link_libraries(http_idle_owners PRIVATE http OpenSSL::SSL)
if(APPLE)
  target_link_libraries(http_idle_owners PRIVATE proc)
endif()
# Private header copies keep the experiment out of the product include tree.
set(small_read_dir "${CMAKE_CURRENT_BINARY_DIR}/small-read")
file(MAKE_DIRECTORY "${small_read_dir}")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../http_asio.h" transport)
string(FIND "${transport}" "char tmp[64 * 1024];" read_buffer)
if(read_buffer EQUAL -1)
  message(FATAL_ERROR "The small-read experiment no longer matches the transport")
endif()
string(REPLACE "char tmp[64 * 1024];" "char tmp[4 * 1024];" transport "${transport}")
file(WRITE "${small_read_dir}/http_asio.h" "${transport}")
configure_file(../../http_asio_tls.h "${small_read_dir}/http_asio_tls.h" COPYONLY)
add_executable(http_idle_small_read server.cc)
target_include_directories(http_idle_small_read BEFORE PRIVATE "${small_read_dir}")
target_link_libraries(http_idle_small_read PRIVATE http OpenSSL::SSL)
get_target_property(http_includes http INTERFACE_INCLUDE_DIRECTORIES)
foreach(directory IN LISTS http_includes)
  if(EXISTS "${directory}/asio/ssl/detail/stream_core.hpp")
    set(idle_asio_include "${directory}")
  endif()
endforeach()
if(NOT idle_asio_include)
  message(FATAL_ERROR "The small-TLS experiment cannot locate pinned Asio")
endif()
set(small_tls_dir "${CMAKE_CURRENT_BINARY_DIR}/small-tls")
file(MAKE_DIRECTORY "${small_tls_dir}/asio/ssl/detail/impl")
file(READ "${idle_asio_include}/asio/ssl/detail/stream_core.hpp" core)
string(FIND "${core}" "max_tls_record_size = 17 * 1024" staging_buffer)
if(staging_buffer EQUAL -1)
  message(FATAL_ERROR "The small-TLS experiment no longer matches Asio staging")
endif()
string(REPLACE "max_tls_record_size = 17 * 1024" "max_tls_record_size = 2 * 1024" core "${core}")
file(WRITE "${small_tls_dir}/asio/ssl/detail/stream_core.hpp" "${core}")
file(READ "${idle_asio_include}/asio/ssl/detail/impl/engine.ipp" engine)
string(FIND "${engine}" "::BIO_new_bio_pair(&int_bio, 0, &ext_bio_, 0);" pair_buffer)
if(pair_buffer EQUAL -1)
  message(FATAL_ERROR "The small-TLS experiment no longer matches Asio BIO construction")
endif()
string(REPLACE "::BIO_new_bio_pair(&int_bio, 0, &ext_bio_, 0);"
  "::BIO_new_bio_pair(&int_bio, 2048, &ext_bio_, 2048);" engine "${engine}")
file(WRITE "${small_tls_dir}/asio/ssl/detail/impl/engine.ipp" "${engine}")
add_executable(http_idle_small_tls server.cc)
target_include_directories(http_idle_small_tls BEFORE PRIVATE "${small_tls_dir}")
target_link_libraries(http_idle_small_tls PRIVATE http OpenSSL::SSL)
add_executable(http_idle_small_both server.cc)
target_include_directories(http_idle_small_both BEFORE PRIVATE "${small_read_dir}" "${small_tls_dir}")
target_link_libraries(http_idle_small_both PRIVATE http OpenSSL::SSL)
add_executable(http_idle_owners_small_tls server.cc)
target_include_directories(http_idle_owners_small_tls BEFORE PRIVATE "${small_tls_dir}")
target_compile_definitions(http_idle_owners_small_tls PRIVATE HTTP_IDLE_SSL_OWNERS=1 TASK_COUNT_FRAMES=1)
target_link_libraries(http_idle_owners_small_tls PRIVATE http OpenSSL::SSL)
if(APPLE)
  target_link_libraries(http_idle_owners_small_tls PRIVATE proc)
endif()
