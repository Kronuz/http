cmake_minimum_required(VERSION 3.20)
project(http_idle_cost LANGUAGES C CXX)
add_subdirectory(../.. http-deps)
find_package(OpenSSL 3 REQUIRED)
add_executable(http_idle_cost server.cc)
target_link_libraries(http_idle_cost PRIVATE http OpenSSL::SSL)
add_executable(http_idle_owners server.cc)
target_compile_definitions(http_idle_owners PRIVATE HTTP_IDLE_SSL_OWNERS=1 TASK_COUNT_FRAMES=1)
target_link_libraries(http_idle_owners PRIVATE http OpenSSL::SSL)
if(APPLE)
  target_link_libraries(http_idle_owners PRIVATE proc)
endif()
# Private header copies keep the experiment out of the product include tree.
set(small_read_dir "${CMAKE_CURRENT_BINARY_DIR}/small-read")
file(MAKE_DIRECTORY "${small_read_dir}")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../http_asio.h" transport)
string(FIND "${transport}" "char tmp[64 * 1024];" read_buffer)
if(read_buffer EQUAL -1)
  message(FATAL_ERROR "The small-read experiment no longer matches the transport")
endif()
string(REPLACE "char tmp[64 * 1024];" "char tmp[4 * 1024];" transport "${transport}")
file(WRITE "${small_read_dir}/http_asio.h" "${transport}")
configure_file(../../http_asio_tls.h "${small_read_dir}/http_asio_tls.h" COPYONLY)
add_executable(http_idle_small_read server.cc)
target_include_directories(http_idle_small_read BEFORE PRIVATE "${small_read_dir}")
target_link_libraries(http_idle_small_read PRIVATE http OpenSSL::SSL)
